function makeFilmstrip(){ const canvas = document.getElementById('film'); if(!canvas) return; const ctx = canvas.getContext('2d'); canvas.width = 300; canvas.height = 60; const dur = vid.duration || 13; let x = 0; const drawFrame = (time) => { vid.currentTime = time; }; vid.addEventListener('seeked', function handler(){ ctx.drawImage(vid, x, 0, 50, 60); x += 52; if(x < 300 && x/52 < 6){ drawFrame((x/52)*(dur/6)); } else { vid.removeEventListener('seeked', handler); vid.currentTime = 0; vid.play(); } }); drawFrame(0.1); } // call it when video loads const oldOnLoaded = vid.onloadeddata; vid.onloadeddata = () => { oldOnLoaded(); setTimeout(makeFilmstrip, 300); };
